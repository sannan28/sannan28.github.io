docker ps                                                   查看所有正在运行的镜像

docker stop 容器id                                    关闭容器

docker inspect 容器ID                               查看容器的运行各种信息  用法示例

docker pull redis                                       下载最新的镜像  镜像的版本需要到docker的官网是去看

docker pull redis:6.2.4                              下载指定的镜像

docker rmi 镜像名|镜像id                           删除镜像
 
docker image rm -f  镜像id                       强制删除镜像

docker ps -a                                              查看所有容器

docker stop 容器名|容器id                         暂停容器

docker start 容器名|容器id                        启动容器
 
docker rm -f 容器id                                    删除容器

docker update 容器名|容器id                    修改容器的启动配置参数

docker inspect 容器id                                查看容器ip              

docker exec -it 容器名|容器id /bin/bash   进入容器内部 并直接进入bash命令 进行命令操作

docker commit -a ‘lsn’ -m ‘test’  镜像id 镜像版本名称  将镜像提交
 -a   作者(auth)
 -m  描述(comment)
